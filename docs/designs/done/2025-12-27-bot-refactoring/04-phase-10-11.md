### Анализ текущего on_message()

```
on_message() делает:
1. Логирование входящего сообщения
2. Пропуск команд (startswith "/")
3. Проверка legacy FSM state (_start_state):
   - awaiting_project_name → вызов _start_project_flow
   - awaiting_custom_path → создание проекта
   - awaiting_clone_url → git clone
4. Роутинг по thread_id:
   - thread есть → отправить в thread tmux
   - thread нет → создать pending thread
   - pending thread → пропустить
5. Session binding (poll_for_session_thread)
6. Отправка в tmux
```

**Проблема:** Legacy FSM (`_start_state`) дублирует aiogram FSM в handlers/start.py.

**Решение:** handlers/start.py уже использует aiogram FSM для start flow.
Нужно удалить legacy FSM из on_message, оставив только роутинг.

#### 10.4 services/message_router.py

```python
from dataclasses import dataclass
from enum import Enum

from ..session_manager import project_manager, ThreadInfo
from ..adapters.tmux import TmuxAdapter

class RouteAction(Enum):
    SEND_TO_TMUX = "send_to_tmux"
    CREATE_PENDING_THREAD = "create_pending_thread"
    SKIP_PENDING = "skip_pending"
    NO_SESSION = "no_session"
    NO_PROJECT = "no_project"
    START_BINDING = "start_binding"

@dataclass
class RouteResult:
    action: RouteAction
    tmux_session: str | None = None
    cwd: str | None = None
    thread: ThreadInfo | None = None
    message: str | None = None

class MessageRouterService:
    """Route incoming messages to appropriate tmux session."""

    def __init__(self, project_manager):
        self.pm = project_manager

    def route(
        self,
        chat_id: int,
        thread_id: int | None,
        text: str,
    ) -> RouteResult:
        """Determine where to route message."""

        project = self.pm.get_by_chat(chat_id)
        if not project:
            return RouteResult(action=RouteAction.NO_PROJECT)

        # Thread routing
        if thread_id is not None:
            thread = project.threads.get(thread_id)

            if not thread:
                # Unknown topic — create pending
                return RouteResult(
                    action=RouteAction.CREATE_PENDING_THREAD,
                    message="Используй /start или /session_new",
                )

            if thread.name == "pending":
                return RouteResult(action=RouteAction.SKIP_PENDING)

            # Has thread — check binding
            if thread.session_id is None:
                return RouteResult(
                    action=RouteAction.START_BINDING,
                    thread=thread,
                )

            tmux_name = thread.get_tmux_session(project.project_name)
            return RouteResult(
                action=RouteAction.SEND_TO_TMUX,
                tmux_session=tmux_name,
                cwd=project.cwd,
                thread=thread,
            )

        # Legacy single-thread mode
        if project.session_id is None:
            return RouteResult(action=RouteAction.START_BINDING)

        if not project.tmux_session:
            return RouteResult(action=RouteAction.NO_SESSION)

        return RouteResult(
            action=RouteAction.SEND_TO_TMUX,
            tmux_session=project.tmux_session,
            cwd=project.cwd,
        )

    def send_to_tmux(self, result: RouteResult, text: str) -> bool:
        """Send text to tmux session."""
        if not result.tmux_session:
            return False

        tmux = TmuxAdapter(result.tmux_session, result.cwd or "/tmp")
        if not tmux.exists():
            return False

        tmux.send(text)
        return True
```

#### 10.5 handlers/messages.py

```python
"""Message routing to tmux sessions."""
from aiogram import Router
from aiogram.types import Message
from aiogram.fsm.context import FSMContext

from ..services.message_router import MessageRouterService, RouteAction
from ..session_manager import project_manager, ThreadInfo
from ..logging_config import logger

router = Router(name="messages")

@router.message()
async def on_message(
    message: Message,
    state: FSMContext,
):
    """Route regular messages to tmux."""

    # Log
    text = message.text or ""
    logger.info(
        f"message: user={message.from_user.id} "
        f"chat={message.chat.id} "
        f"thread={message.message_thread_id}: "
        f"{text[:50]}"
    )

    if not text:
        return

    # Skip commands
    if text.startswith("/"):
        return

    # Check FSM state — if in flow, aiogram handles it
    current_state = await state.get_state()
    if current_state is not None:
        # FSM handler will process this
        return

    # Route message
    result = message_router.route(
        chat_id=message.chat.id,
        thread_id=message.message_thread_id,
        text=text,
    )

    match result.action:
        case RouteAction.NO_PROJECT:
            # Silent — no project registered
            return

        case RouteAction.CREATE_PENDING_THREAD:
            thread = ThreadInfo(
                thread_id=message.message_thread_id,
                name="pending",
            )
            project = project_manager.get_by_chat(message.chat.id)
            project.threads[message.message_thread_id] = thread
            project_manager._save()
            await message.answer(result.message)
            return

        case RouteAction.SKIP_PENDING:
            return

        case RouteAction.START_BINDING:
            await _start_session_binding(message, result)
            # Still try to send to tmux
            if result.tmux_session:
                message_router.send_to_tmux(result, text)
            return

        case RouteAction.NO_SESSION:
            if message.chat.id < 0:  # Group
                await message.answer("Нет активной сессии. /start")
            return

        case RouteAction.SEND_TO_TMUX:
            success = message_router.send_to_tmux(result, text)
            if not success:
                logger.warning(f"tmux send failed: {result.tmux_session}")
                if message.chat.id < 0:
                    await message.answer("Нет активной сессии. /start")

async def _start_session_binding(message: Message, result):
    """Start polling for session binding."""
    import asyncio
    from ..history_watcher import poll_for_session_thread
    from .. import main

    project = project_manager.get_by_chat(message.chat.id)
    thread = result.thread

    if not thread:
        return  # No thread to bind

    thread.last_sent_message = message.text

    if not thread.binding_task or thread.binding_task.done():
        # Create closure for start_poller/start_watcher
        async def start_poller(p):
            from ..permission_poller import create_poller_task
            return await create_poller_task(message.bot, p, main.telegram_queue)

        async def start_watcher(p, send_missed=False):
            from ..watcher import create_watcher_task
            return await create_watcher_task(message.bot, p, main.telegram_queue, send_missed)

        thread.binding_task = asyncio.create_task(
            poll_for_session_thread(
                project, thread, message.bot,
                start_poller, start_watcher, main.telegram_queue
            )
        )
```

```python
"""Handlers layer - thin routers delegating to services."""
from aiogram import Dispatcher

from . import permissions, start, threads, branches, sessions, settings, common, messages


def register_handlers(dp: Dispatcher):
    """Register all handler routers.

    Note: AdminMiddleware is registered on dp level in main.py,
    protecting ALL routers. No need to add it here.

    Order matters:
    - common.router has cb_cancel (generic cancel)
    - messages.router is catch-all (must be last)
    """
    dp.include_router(permissions.router)   # Permission callbacks
    dp.include_router(start.router)         # /start, /restart
    dp.include_router(threads.router)       # /thread_create, /thread_delete
    dp.include_router(branches.router)      # /branch_create, /branch_finish
    dp.include_router(sessions.router)      # /new, /clear, /esc, /resume
    dp.include_router(settings.router)      # /settings, /auto_accept, /help
    dp.include_router(common.router)        # cb_cancel
    dp.include_router(messages.router)      # Catch-all for tmux routing
```
