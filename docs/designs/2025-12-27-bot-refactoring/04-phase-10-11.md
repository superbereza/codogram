## Фаза 10: Вынести handlers/messages.py

**Цель:** Последний и самый сложный handler — роутинг сообщений в tmux

### Анализ текущего on_message()

```
on_message() делает:
1. Логирование входящего сообщения
2. Проверка admin (уберём — middleware)
3. Пропуск команд (startswith "/")
4. Проверка FSM state (_start_state) — теперь aiogram FSM
5. Роутинг по thread_id:
   - thread есть → отправить в thread tmux
   - thread нет → создать pending thread
   - pending thread → показать hint
6. Session binding (poll_for_session)
7. Отправка в tmux
```

#### 10.1 services/message_router.py

```python
from dataclasses import dataclass
from enum import Enum

from ..session_manager import project_manager, ThreadInfo
from ..adapters.tmux import TmuxAdapter

class RouteAction(Enum):
    SEND_TO_TMUX = "send_to_tmux"
    CREATE_PENDING_THREAD = "create_pending_thread"
    SKIP_PENDING = "skip_pending"
    NO_SESSION = "no_session"
    NO_PROJECT = "no_project"
    START_BINDING = "start_binding"

@dataclass
class RouteResult:
    action: RouteAction
    tmux_session: str | None = None
    cwd: str | None = None
    thread: ThreadInfo | None = None
    message: str | None = None

class MessageRouterService:
    """Route incoming messages to appropriate tmux session."""

    def __init__(self, project_manager):
        self.pm = project_manager

    def route(
        self,
        chat_id: int,
        thread_id: int | None,
        text: str,
    ) -> RouteResult:
        """Determine where to route message."""

        project = self.pm.get_by_chat(chat_id)
        if not project:
            return RouteResult(action=RouteAction.NO_PROJECT)

        # Thread routing
        if thread_id is not None:
            thread = project.threads.get(thread_id)

            if not thread:
                # Unknown topic — create pending
                return RouteResult(
                    action=RouteAction.CREATE_PENDING_THREAD,
                    message="Используй /start или /session_new",
                )

            if thread.name == "pending":
                return RouteResult(action=RouteAction.SKIP_PENDING)

            # Has thread — check binding
            if thread.session_id is None:
                return RouteResult(
                    action=RouteAction.START_BINDING,
                    thread=thread,
                )

            tmux_name = thread.get_tmux_session(project.project_name)
            return RouteResult(
                action=RouteAction.SEND_TO_TMUX,
                tmux_session=tmux_name,
                cwd=project.cwd,
                thread=thread,
            )

        # Legacy single-thread mode
        if project.session_id is None:
            return RouteResult(action=RouteAction.START_BINDING)

        if not project.tmux_session:
            return RouteResult(action=RouteAction.NO_SESSION)

        return RouteResult(
            action=RouteAction.SEND_TO_TMUX,
            tmux_session=project.tmux_session,
            cwd=project.cwd,
        )

    def send_to_tmux(self, result: RouteResult, text: str) -> bool:
        """Send text to tmux session."""
        if not result.tmux_session:
            return False

        tmux = TmuxAdapter(result.tmux_session, result.cwd or "/tmp")
        if not tmux.exists():
            return False

        tmux.send(text)
        return True
```

#### 10.2 handlers/messages.py

```python
from aiogram import Router
from aiogram.types import Message
from aiogram.fsm.context import FSMContext

from ..services.message_router import MessageRouterService, RouteAction
from ..session_manager import project_manager, ThreadInfo
from ..logging_config import logger

router = Router()

@router.message()
async def on_message(
    message: Message,
    state: FSMContext,
    message_router: MessageRouterService,
):
    """Route regular messages to tmux."""

    # Log
    text = message.text or ""
    logger.info(
        f"message: user={message.from_user.id} "
        f"chat={message.chat.id} "
        f"thread={message.message_thread_id}: "
        f"{text[:50]}"
    )

    if not text:
        return

    # Skip commands
    if text.startswith("/"):
        return

    # Check FSM state — if in flow, aiogram handles it
    current_state = await state.get_state()
    if current_state is not None:
        # FSM handler will process this
        return

    # Route message
    result = message_router.route(
        chat_id=message.chat.id,
        thread_id=message.message_thread_id,
        text=text,
    )

    match result.action:
        case RouteAction.NO_PROJECT:
            # Silent — no project registered
            return

        case RouteAction.CREATE_PENDING_THREAD:
            thread = ThreadInfo(
                thread_id=message.message_thread_id,
                name="pending",
            )
            project = project_manager.get_by_chat(message.chat.id)
            project.threads[message.message_thread_id] = thread
            project_manager._save()
            await message.answer(result.message)
            return

        case RouteAction.SKIP_PENDING:
            return

        case RouteAction.START_BINDING:
            await _start_session_binding(message, result)
            # Still try to send to tmux
            if result.tmux_session:
                message_router.send_to_tmux(result, text)
            return

        case RouteAction.NO_SESSION:
            if message.chat.id < 0:  # Group
                await message.answer("Нет активной сессии. /start")
            return

        case RouteAction.SEND_TO_TMUX:
            success = message_router.send_to_tmux(result, text)
            if not success:
                logger.warning(f"tmux send failed: {result.tmux_session}")
                if message.chat.id < 0:
                    await message.answer("Нет активной сессии. /start")
```

#### 10.3 handlers/__init__.py (финальный)

```python
from aiogram import Dispatcher, Router
from . import permissions, start, sessions, messages, public
from ..middleware.admin import AdminMiddleware
from ..middleware.dependencies import DependencyMiddleware

def register_handlers(
    dp: Dispatcher,
    launch_service,
    start_flow_service,
    message_router_service,
):
    # Dependency injection middleware
    dep_middleware = DependencyMiddleware(
        launch_service=launch_service,
        start_flow_service=start_flow_service,
        message_router=message_router_service,
    )

    # Public handlers
    dp.include_router(public.router)

    # Admin handlers
    admin_router = Router()
    admin_router.message.middleware(AdminMiddleware())
    admin_router.callback_query.middleware(AdminMiddleware())
    admin_router.message.middleware(dep_middleware)
    admin_router.callback_query.middleware(dep_middleware)

    admin_router.include_router(start.router)
    admin_router.include_router(sessions.router)
    admin_router.include_router(permissions.router)
    admin_router.include_router(messages.router)  # Last — catch-all

    dp.include_router(admin_router)
```

```python
import asyncio
from aiogram import Bot, Dispatcher
from aiogram.fsm.storage.memory import MemoryStorage

from .config import settings, load_config
from .session_manager import project_manager
from .handlers import register_handlers
from .services.launch import LaunchService
from .services.start_flow import StartFlowService
from .services.message_router import MessageRouterService
from .adapters.telegram import TelegramAdapter
from .logging_config import logger

async def main():
    # Load config
    config = load_config()

    # Create bot and dispatcher
    bot = Bot(token=settings.TELEGRAM_TOKEN)
    storage = MemoryStorage()  # Or Redis for persistence
    dp = Dispatcher(storage=storage)

    # Create services
    telegram_adapter = TelegramAdapter()
    launch_service = LaunchService(project_manager, telegram_adapter)
    start_flow_service = StartFlowService(project_manager, launch_service)
    message_router = MessageRouterService(project_manager)

    # Register handlers
    register_handlers(
        dp,
        launch_service=launch_service,
        start_flow_service=start_flow_service,
        message_router_service=message_router,
    )

    # Restore projects from config
    await _restore_projects(bot, launch_service)

    # Start polling
    logger.info("Starting bot...")
    await dp.start_polling(bot)

async def _restore_projects(bot: Bot, launch_service: LaunchService):
    """Restore active projects on startup."""
    for project in project_manager.get_all_active():
        try:
            await launch_service.restore_project(project, bot)
        except Exception as e:
            logger.error(f"Failed to restore {project.project_name}: {e}")

if __name__ == "__main__":
    asyncio.run(main())
```
